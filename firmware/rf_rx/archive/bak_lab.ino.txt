/*
 * ============================================================
 *  Laboratorio de controles remotos 433 MHz (ESP32 + CC1101)
 * ============================================================
 *
 *  IDEIA GERAL
 *  Temos uma LISTA DE EXPERIMENTOS (modulacoes e configuracoes
 *  diferentes). Cada clique no botao passa para o proximo
 *  experimento e reconfigura o radio. O Serial Monitor (115200)
 *  mostra qual esta ativo e o que foi capturado.
 *
 *  DOIS JEITOS DE RECEBER
 *  - CRU: o CC1101 so entrega o sinal (liga/desliga) no pino GDO0
 *    e o ESP32 mede os tempos. Funciona com qualquer controle,
 *    mas quem interpreta somos nos.
 *  - PACOTE: o proprio CC1101 procura um "sync word", junta os
 *    bits em bytes e entrega pronto. So funciona se acertarmos
 *    toda a configuracao do transmissor.
 *
 *  LIGACAO
 *    CC1101 -> ESP32            Botao -> ESP32
 *    1 GND  -> GND              perna   -> D22
 *    2 VCC  -> 3V3 (nunca 5V!)  diagonal-> GND
 *    3 GDO0 -> D4  (GPIO 4) 
 *    4 CSN  -> D5
 *    5 SCK  -> D18
 *    6 MOSI -> D23
 *    7 MISO -> D19
 *    8 GDO2 -> D2  (GPIO 2, nao usado neste codigo)
 *
 *  BIBLIOTECA: "CC1101" (mfurga)
 */

#include <SPI.h>
#include <cc1101.h>

using namespace CC1101;


// ============================================================
//  PINOS
// ============================================================
#define PIN_SCK    18
#define PIN_MISO   19
#define PIN_MOSI   23
#define PIN_CSN     5
#define PIN_GDO0    4
#define PIN_BOTAO  22


// ============================================================
//  LISTA DE EXPERIMENTOS (acrescente linhas a vontade)
// ============================================================
enum Tipo { OOK_CRU, FSK_CRU, PACOTE };

struct Experimento {
  const char* nome;
  Tipo     tipo;
  float    taxaKbaud;   // taxa de dados
  float    desvioKhz;   // so FSK: distancia entre as duas frequencias
  uint16_t sync;        // so PACOTE: palavra de sincronismo
  bool     fec;         // so PACOTE: correcao de erros
};

Experimento experimentos[] = {
  // nome                                tipo     taxa   desvio  sync    fec
  { "OOK cru",                           OOK_CRU, 100.0,  0.0,   0,      false },
  { "FSK cru, desvio 25 kHz",            FSK_CRU, 100.0, 25.0,   0,      false },
  { "FSK cru, desvio 50 kHz",            FSK_CRU, 100.0, 50.0,   0,      false },
  { "Pacote FSK 4.8k, sync D391, FEC",   PACOTE,    4.8, 25.0,   0xD391, true  },
  { "Pacote FSK 4.8k, sync D391",        PACOTE,    4.8, 25.0,   0xD391, false },
  { "Pacote FSK 2.4k, sync D391, FEC",   PACOTE,    2.4, 25.0,   0xD391, true  },
  { "Pacote FSK 38.4k, sync D391, FEC",  PACOTE,   38.4, 20.0,   0xD391, true  },
};

const int TOTAL_EXPERIMENTOS = sizeof(experimentos) / sizeof(experimentos[0]);
int atual = 0;   // qual experimento esta ativo


// ============================================================
//  AJUSTES GERAIS
// ============================================================
const float FREQUENCIA_MHZ = 433.92;
const int   LIMITE_DBM     = -55;             // modos crus: acima disso = controle apertado
const unsigned long TEMPO_CAPTURA_MS = 3000;  // modos crus: quanto tempo gravar
const bool  CONGELAR_AGC   = true;            // modos crus: fixa o ganho durante a captura
const size_t TAMANHO_PACOTE = 8;              // modo pacote: bytes por pacote (FEC exige fixo)

// Tempos de um bit no OOK (microssegundos):
//   0 = ligado curto + desligado longo,  1 = ligado longo + desligado curto
const long CURTO_MIN = 150,  CURTO_MAX = 450;
const long LONGO_MIN = 550,  LONGO_MAX = 1000;
const long PAUSA_MIN = 5000;      // pausa = fim de bloco
const long PAUSA_QUADRO = 11800;  // pausa maior = fim de quadro


// ============================================================
//  REGISTRADORES DO CC1101
// ============================================================
const uint8_t END_IOCFG0   = 0x02;  // o que sai no pino GDO0
const uint8_t END_PKTCTRL0 = 0x08;  // formato dos dados
const uint8_t END_AGCCTRL2 = 0x1B;  // ganho automatico (AGC)
const uint8_t END_AGCCTRL1 = 0x1C;
const uint8_t END_AGCCTRL0 = 0x1D;
const uint8_t END_FREND1   = 0x21;  // estagio de recepcao
const uint8_t END_RSSI      = 0x34; // (status) forca do sinal
const uint8_t END_MARCSTATE = 0x35; // (status) estado do chip, 13 = recebendo
const uint8_t CMD_ENTRAR_RX = 0x34; // comando: comece a escutar

Radio radio(PIN_CSN, PIN_GDO0);


// ============================================================
//  LEITURA DE STATUS (SPI "puro")
// ============================================================
uint8_t lerStatus(uint8_t endereco) {
  SPI.beginTransaction(SPISettings(1000000, MSBFIRST, SPI_MODE0));
  digitalWrite(PIN_CSN, LOW);                  // chama o CC1101
  while (digitalRead(PIN_MISO) == HIGH);       // espera ficar pronto
  SPI.transfer(endereco | 0xC0);               // pede registrador de status
  uint8_t valor = SPI.transfer(0x00);          // recebe a resposta
  digitalWrite(PIN_CSN, HIGH);
  SPI.endTransaction();
  return valor;
}

int lerSinalDbm() {
  int bruto = lerStatus(END_RSSI);
  if (bruto >= 128) bruto -= 256;
  return bruto / 2 - 74;
}

void imprimirTempo() {
  Serial.print("[");
  Serial.print(millis() / 1000.0, 3);
  Serial.print("s] ");
}


// ============================================================
//  CONFIGURACAO DO RADIO PARA O EXPERIMENTO ATUAL
// ============================================================
void configurarExperimento() {
  Experimento& e = experimentos[atual];

  radio.idle();   // para o radio antes de mudar qualquer coisa

  // Comum a todos
  radio.setFrequency(FREQUENCIA_MHZ);
  radio.setRxBandwidth(270.0);
  radio.setDataRate(e.taxaKbaud);

  if (e.tipo == OOK_CRU) {
    radio.setModulation(MOD_ASK_OOK);       // liga/desliga a portadora
  } else {
    radio.setModulation(MOD_2FSK);          // alterna entre duas frequencias
    radio.setFrequencyDeviation(e.desvioKhz);
  }

  if (e.tipo == PACOTE) {
    // O CC1101 monta os pacotes sozinho
    radio.writeReg(END_AGCCTRL1, 0x40);     // AGC no padrao de fabrica
    radio.setPacketFormat(PKT_FORMAT_NORMAL);
    radio.setPacketLengthMode(PKT_LEN_MODE_FIXED, TAMANHO_PACOTE);
    radio.setSyncWord(e.sync);
    radio.setSyncMode(SYNC_MODE_15_16);     // tolera 1 bit errado no sync
    radio.setCrc(false);                    // sem CRC: queremos ver qualquer coisa
    radio.setDataWhitening(false);
    radio.setManchester(false);
    radio.setFEC(e.fec);
    return;                                 // radio.receive() liga o RX sozinho
  }

  // Modos crus: desliga o montador de pacotes
  radio.setPacketLengthMode(PKT_LEN_MODE_INFINITE);
  radio.setSyncMode(SYNC_MODE_NO_PREAMBLE);
  radio.setCrc(false);
  radio.setDataWhitening(false);
  radio.setManchester(false);
  radio.setFEC(false);
  radio.setPacketFormat(PKT_FORMAT_ASYNC_SERIAL);

  radio.writeReg(END_IOCFG0,   0x0D);       // GDO0 entrega o sinal cru
  radio.writeReg(END_PKTCTRL0, 0x32);       // serial assincrono, sem pacotes
  radio.writeReg(END_AGCCTRL2, 0x03);       // sensibilidade recomendada pela TI
  radio.writeReg(END_AGCCTRL1, 0x00);
  radio.writeReg(END_AGCCTRL0, 0x91);
  radio.writeReg(END_FREND1,   0xB6);

  radio.sendCmd(CMD_ENTRAR_RX);             // liga o receptor
  delay(10);                                // tempo para calibrar
  pinMode(PIN_GDO0, INPUT);
}

void anunciarExperimento() {
  Serial.println();
  Serial.println("==================================================");
  Serial.print("Experimento ");
  Serial.print(atual + 1);
  Serial.print(" de ");
  Serial.print(TOTAL_EXPERIMENTOS);
  Serial.print(": ");
  Serial.println(experimentos[atual].nome);
  Serial.println("Aperte e segure o controle. Botao = proximo teste.");
  Serial.println("==================================================");
}


// ============================================================
//  BOTAO: detecta um clique (com filtro de "bouncing")
// ============================================================
bool botaoClicado() {
  static int estadoAnterior = HIGH;
  static unsigned long ultimaMudanca = 0;

  int estado = digitalRead(PIN_BOTAO);     // INPUT_PULLUP: solto = HIGH
  if (estado != estadoAnterior && millis() - ultimaMudanca > 50) {
    ultimaMudanca = millis();
    estadoAnterior = estado;
    return estado == LOW;                  // conta so a descida (apertou)
  }
  return false;
}


// ============================================================
//  CAPTURA CRUA 1: PULSOS EM MICROSSEGUNDOS
// ============================================================
// Positivo = ligado (ou freq. alta no FSK), negativo = desligado
void capturaPulsos() {
  const int MAX = 400;
  long duracao[MAX];
  int n = 0;

  int nivelAnterior = digitalRead(PIN_GDO0);
  unsigned long ultimo = micros();
  unsigned long fim = millis() + TEMPO_CAPTURA_MS;

  while (millis() < fim && n < MAX) {
    int nivel = digitalRead(PIN_GDO0);
    if (nivel == nivelAnterior) continue;
    unsigned long agora = micros();
    long d = agora - ultimo;
    duracao[n++] = nivelAnterior ? d : -d;
    ultimo = agora;
    nivelAnterior = nivel;
  }

  Serial.print("Pulsos: ");
  Serial.println(n);
  for (int i = 0; i < n; i++) {
    Serial.print(duracao[i]);
    Serial.print((i + 1) % 10 == 0 ? "\n" : " ");
  }
  Serial.println();
}


// ============================================================
//  CAPTURA CRUA 2: BITS (so faz sentido no OOK)
// ============================================================
// O bit e decidido pelo tempo DESLIGADO, que chega sempre certo.
// O ultimo bit de cada bloco usa o tempo ligado; se duvidoso, "?".
void capturaBits() {
  String bloco = "", saida = "";
  long ligado = 0;
  int nivel = digitalRead(PIN_GDO0);
  unsigned long t0 = micros();
  unsigned long fim = millis() + TEMPO_CAPTURA_MS;

  while (millis() < fim) {
    int novoNivel = digitalRead(PIN_GDO0);
    if (novoNivel == nivel) continue;
    unsigned long agora = micros();
    long d = agora - t0;
    t0 = agora;

    if (nivel == HIGH) {
      ligado = d;
    } else {
      if (d > PAUSA_MIN) {
        if (ligado > LONGO_MIN && ligado < LONGO_MAX) bloco += "1";
        else if (ligado > 200 && ligado < 320)        bloco += "0";
        else if (ligado > 0)                          bloco += "?";
        if (bloco.length() >= 4) saida += bloco + "(" + String(d / 1000) + "ms) ";
        if (d > PAUSA_QUADRO) saida += "\n    ";
        bloco = "";
      } else if (d > CURTO_MIN && d < CURTO_MAX) {
        bloco += "1";
      } else if (d > LONGO_MIN && d < LONGO_MAX) {
        bloco += "0";
      }
      ligado = 0;
    }
    nivel = novoNivel;
  }
  Serial.println(saida);
}


// ============================================================
//  ESCUTA NOS MODOS CRUS
// ============================================================
void escutarCru() {
  int dbm = lerSinalDbm();
  if (dbm <= LIMITE_DBM) return;

  imprimirTempo();
  Serial.print("Sinal forte (");
  Serial.print(dbm);
  Serial.print(" dBm): ");

  if (CONGELAR_AGC) radio.writeReg(END_AGCCTRL0, 0x9D);  // congela o ganho

  if (experimentos[atual].tipo == OOK_CRU) capturaBits();
  else                                     capturaPulsos();

  if (CONGELAR_AGC) radio.writeReg(END_AGCCTRL0, 0x91);  // descongela

  delay(1000);
}


// ============================================================
//  ESCUTA NO MODO PACOTE
// ============================================================
void escutarPacote() {
  uint8_t dados[TAMANHO_PACOTE];
  size_t lidos = 0;

  // Espera um pacote (com tempo limite). Se nao vier nada, sai.
  Status st = radio.receive(dados, TAMANHO_PACOTE, &lidos);
  if (st != STATUS_OK) return;

  imprimirTempo();
  Serial.print("PACOTE (");
  Serial.print(radio.getRSSI());
  Serial.print(" dBm): ");
  for (size_t i = 0; i < lidos; i++) {
    if (dados[i] < 0x10) Serial.print('0');
    Serial.print(dados[i], HEX);
    Serial.print(' ');
  }
  Serial.println();
}


// ============================================================
//  SETUP
// ============================================================
void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(PIN_BOTAO, INPUT_PULLUP);
  pinMode(PIN_CSN, OUTPUT);
  digitalWrite(PIN_CSN, HIGH);
  SPI.begin(PIN_SCK, PIN_MISO, PIN_MOSI, PIN_CSN);

  if (radio.begin() == STATUS_CHIP_NOT_FOUND) {
    Serial.println("CC1101 nao encontrado. Confira os fios.");
    while (true) delay(1000);
  }
  Serial.println("CC1101 encontrado!");

  configurarExperimento();
  anunciarExperimento();
}


// ============================================================
//  LOOP
// ============================================================
void loop() {
  if (botaoClicado()) {
    atual = (atual + 1) % TOTAL_EXPERIMENTOS;   // proximo (volta ao primeiro no fim)
    configurarExperimento();
    anunciarExperimento();
  }

  if (experimentos[atual].tipo == PACOTE) escutarPacote();
  else                                    escutarCru();
}
