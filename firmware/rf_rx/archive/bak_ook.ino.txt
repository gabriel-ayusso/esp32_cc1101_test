/*
 * ============================================================
 *  Leitor de controles remotos 433 MHz (ESP32 + CC1101)
 * ============================================================
 *
 *  COMO FUNCIONA, EM 3 PARTES:
 *
 *  1. O CC1101 e o "ouvido": sintoniza 433,92 MHz e entrega o
 *     sinal cru (liga/desliga) no pino GDO0.
 *
 *  2. O ESP32 fica medindo a FORCA do sinal (RSSI). Quando
 *     alguem aperta um controle perto, o sinal sobe muito.
 *
 *  3. Nesse momento, o ESP32 grava os pulsos do GDO0 e mostra
 *     no Serial Monitor (115200), em bits ou em microssegundos.
 *
 *  LIGACAO (CC1101 -> ESP32):
 *     1 GND  -> GND
 *     2 VCC  -> 3V3   (nunca 5V!)
 *     3 GDO0 -> RX2   (GPIO 16)
 *     4 CSN  -> D5
 *     5 SCK  -> D18
 *     6 MOSI -> D23
 *     7 MISO -> D19
 *     8 GDO2 -> TX2   (GPIO 17, nao usado aqui)
 *
 *  BIBLIOTECA: "CC1101" (mfurga), instalada pelo Gerenciador.
 */

#include <SPI.h>
#include <cc1101.h>

using namespace CC1101;


// ============================================================
//  PINOS
// ============================================================
#define PIN_SCK   18
#define PIN_MISO  19
#define PIN_MOSI  23
#define PIN_CSN    5
#define PIN_GDO0  16


// ============================================================
//  AJUSTES (mexa aqui a vontade)
// ============================================================
const float FREQUENCIA_MHZ   = 433.92;  // frequencia dos controles
const int   LIMITE_DBM       = -55;     // acima disso, consideramos "controle apertado"
const unsigned long TEMPO_CAPTURA_MS = 3000;  // quanto tempo gravar a cada clique

// true  = mostra os bits traduzidos (0 e 1)
// false = mostra os pulsos crus em microssegundos
const bool MOSTRAR_BITS = true;

// true  = desliga o ajuste automatico de ganho (AGC): o ganho fica fixo
// false = o CC1101 reajusta o ganho sozinho o tempo todo (padrao)
const bool CONGELAR_AGC = true;

// Tempos que definem um bit (em microssegundos).
// Esse controle usa pulsos de ~250 us (curto) e ~760 us (longo):
//   curto ligado + longo desligado = 0
//   longo ligado + curto desligado = 1
const long CURTO_MIN = 150,  CURTO_MAX = 450;
const long LONGO_MIN = 550,  LONGO_MAX = 1000;
const long PAUSA_MIN = 5000; // desligado por mais que isso = fim de um bloco
const long PAUSA_QUADRO = 11800; // pausa maior (~12 ms) = fim de um quadro completo


// ============================================================
//  REGISTRADORES DO CC1101 (as "gavetinhas" de configuracao)
// ============================================================
// Configuracao
const uint8_t END_IOCFG0   = 0x02;  // o que sai no pino GDO0
const uint8_t END_PKTCTRL0 = 0x08;  // como o chip trata os dados
const uint8_t END_AGCCTRL2 = 0x1B;  // controle automatico de ganho (sensibilidade)
const uint8_t END_AGCCTRL1 = 0x1C;
const uint8_t END_AGCCTRL0 = 0x1D;
const uint8_t END_FREND1   = 0x21;  // ajuste do estagio de recepcao

// Status (so leitura)
const uint8_t END_RSSI      = 0x34; // forca do sinal
const uint8_t END_MARCSTATE = 0x35; // o que o chip esta fazendo (13 = recebendo)

// Comandos
const uint8_t CMD_ENTRAR_RX = 0x34; // "comece a escutar"


Radio radio(PIN_CSN, PIN_GDO0);


// ============================================================
//  LEITURA DE STATUS VIA SPI "PURO"
// ============================================================
// Registradores de status precisam de um bit especial na leitura
// (0xC0). Fazemos na mao para garantir que funcione.
uint8_t lerStatus(uint8_t endereco) {
  SPI.beginTransaction(SPISettings(1000000, MSBFIRST, SPI_MODE0));
  digitalWrite(PIN_CSN, LOW);                  // 1. chama o CC1101
  while (digitalRead(PIN_MISO) == HIGH);       // 2. espera ele ficar pronto
  SPI.transfer(endereco | 0xC0);               // 3. pede o registrador
  uint8_t valor = SPI.transfer(0x00);          // 4. recebe a resposta
  digitalWrite(PIN_CSN, HIGH);                 // libera o CC1101
  SPI.endTransaction();
  return valor;
}

// Converte o valor bruto do RSSI para dBm (formula do datasheet)
int lerSinalDbm() {
  int bruto = lerStatus(END_RSSI);
  if (bruto >= 128) bruto -= 256;
  return bruto / 2 - 74;
}


// ============================================================
//  CONFIGURACAO DO RADIO
// ============================================================
void configurarRadio() {
  // Sintonia: como os controles remotos transmitem
  radio.setModulation(MOD_ASK_OOK);       // OOK = liga/desliga a portadora
  radio.setFrequency(FREQUENCIA_MHZ);
  radio.setRxBandwidth(270.0);            // janela larga: controles baratos variam
  radio.setDataRate(100);                 // amostragem alta para o modo cru

  // Desliga o "montador de pacotes": queremos so o sinal cru
  radio.setPacketLengthMode(PKT_LEN_MODE_INFINITE);
  radio.setSyncMode(SYNC_MODE_NO_PREAMBLE);
  radio.setCrc(false);
  radio.setDataWhitening(false);
  radio.setManchester(false);
  radio.setFEC(false);
  radio.setPacketFormat(PKT_FORMAT_ASYNC_SERIAL);  // avisa a biblioteca do modo cru

  // Modo serial assincrono, configurado direto nos registradores
  radio.writeReg(END_IOCFG0,   0x0D);     // GDO0 entrega o sinal demodulado cru
  radio.writeReg(END_PKTCTRL0, 0x32);     // serial assincrono, tamanho infinito, sem CRC

  // Sensibilidade recomendada pela TI para OOK
  radio.writeReg(END_AGCCTRL2, 0x03);
  radio.writeReg(END_AGCCTRL1, 0x00);
  radio.writeReg(END_AGCCTRL0, 0x91);

  radio.writeReg(END_FREND1,   0xB6);

  // Liga o receptor
  radio.sendCmd(CMD_ENTRAR_RX);
  delay(10);                              // tempo para o chip calibrar e entrar em RX
  pinMode(PIN_GDO0, INPUT);
}


// ============================================================
//  CAPTURA 1: PULSOS CRUS
// ============================================================
// Mostra a duracao de cada pulso em microssegundos.
// Positivo = radio ligado, negativo = desligado.
void capturaPulsos() {
  const int MAX = 400;
  long duracao[MAX];
  int n = 0;

  int nivelAnterior = digitalRead(PIN_GDO0);
  unsigned long ultimo = micros();
  unsigned long fim = millis() + TEMPO_CAPTURA_MS;

  while (millis() < fim && n < MAX) {
    int nivel = digitalRead(PIN_GDO0);
    if (nivel == nivelAnterior) continue;

    unsigned long agora = micros();
    long d = agora - ultimo;
    duracao[n++] = nivelAnterior ? d : -d;
    ultimo = agora;
    nivelAnterior = nivel;
  }

  Serial.print("Pulsos: ");
  Serial.println(n);
  for (int i = 0; i < n; i++) {
    Serial.print(duracao[i]);
    Serial.print((i + 1) % 10 == 0 ? "\n" : " ");
  }
  Serial.println();
}


// ============================================================
//  CAPTURA 2: BITS TRADUZIDOS
// ============================================================
// Cada bit e um par (ligado, desligado):
//   0 = ligado CURTO + desligado LONGO
//   1 = ligado LONGO + desligado CURTO
//
// Descobrimos que o PRIMEIRO e o ULTIMO pulso ligado de cada bloco
// chegam encurtados (o receptor se reajusta depois do silencio).
// Por isso decidimos o bit olhando o tempo DESLIGADO, que chega
// sempre certinho. So no ultimo bit do bloco, que termina numa
// pausa, usamos o tempo ligado; se ele estiver duvidoso, marcamos "?".
//
// Saida: cada bloco seguido da pausa depois dele, ex: 1101110(10ms)
void capturaBits() {
  String bloco = "";
  String saida = "";
  long ligado = 0;

  int nivel = digitalRead(PIN_GDO0);
  unsigned long t0 = micros();
  unsigned long fim = millis() + TEMPO_CAPTURA_MS;

  while (millis() < fim) {
    int novoNivel = digitalRead(PIN_GDO0);
    if (novoNivel == nivel) continue;

    unsigned long agora = micros();
    long d = agora - t0;
    t0 = agora;

    if (nivel == HIGH) {
      // Acabou um pulso LIGADO: guarda a duracao
      ligado = d;
    } else {
      // Acabou um pulso DESLIGADO: decide o bit pelo tempo desligado
      if (d > PAUSA_MIN) {
        // Pausa longa: ultimo bit do bloco, decide pelo tempo ligado
        if (ligado > LONGO_MIN && ligado < LONGO_MAX)      bloco += "1";
        else if (ligado > 200 && ligado < 320)             bloco += "0";
        else if (ligado > 0)                               bloco += "?";

        if (bloco.length() >= 4) {
          saida += bloco + "(" + String(d / 1000) + "ms) ";
        }
        if (d > PAUSA_QUADRO) saida += "\n    ";  // fim de quadro: nova linha
        bloco = "";
      } else if (d > CURTO_MIN && d < CURTO_MAX) {
        bloco += "1";   // desligado curto: era um ligado longo
      } else if (d > LONGO_MIN && d < LONGO_MAX) {
        bloco += "0";   // desligado longo: era um ligado curto
      }
      // Qualquer outra duracao e ruido: ignoramos

      ligado = 0;
    }
    nivel = novoNivel;
  }

  Serial.println(saida);
}


// ============================================================
//  SETUP: roda uma vez ao ligar
// ============================================================
void setup() {
  Serial.begin(115200);
  delay(1000);

  pinMode(PIN_CSN, OUTPUT);
  digitalWrite(PIN_CSN, HIGH);
  SPI.begin(PIN_SCK, PIN_MISO, PIN_MOSI, PIN_CSN);

  if (radio.begin() == STATUS_CHIP_NOT_FOUND) {
    Serial.println("CC1101 nao encontrado. Confira os fios.");
    while (true) delay(1000);
  }
  Serial.println("CC1101 encontrado!");

  configurarRadio();

  uint8_t estado = lerStatus(END_MARCSTATE) & 0x1F;
  Serial.print("Estado do radio: ");
  Serial.print(estado);
  Serial.println(estado == 13 ? " (recebendo)" : " (ATENCAO: nao esta recebendo)");

  Serial.println("Pronto! Aperte um controle perto da antena.\n");
}


// ============================================================
//  LOOP: roda sem parar
// ============================================================
void loop() {
  int dbm = lerSinalDbm();

  if (dbm > LIMITE_DBM) {
    // Tempo desde que a placa ligou, para saber quando foi cada clique
    Serial.print("[");
    Serial.print(millis() / 1000.0, 3);
    Serial.print("s] Sinal forte (");
    Serial.print(dbm);
    Serial.print(" dBm): ");

    // Congela o ganho AGORA, que ele ja se ajustou ao sinal do controle.
    // AGCCTRL0, bits 3-2 (AGC_FREEZE): 00 = automatico, 11 = congelado.
    // 0x91 vira 0x9D quando ligamos esses dois bits.
    if (CONGELAR_AGC) radio.writeReg(END_AGCCTRL0, 0x9D);

    if (MOSTRAR_BITS) capturaBits();
    else              capturaPulsos();

    // Descongela para voltar a se adaptar ao proximo sinal
    if (CONGELAR_AGC) radio.writeReg(END_AGCCTRL0, 0x91);

    delay(1500);  // pausa para nao capturar o mesmo clique duas vezes
  }
}
